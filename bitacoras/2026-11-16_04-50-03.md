### Opción 1: Desde la interfaz de Android Studio (La más fácil y rápida)
1. En la barra superior de Android Studio ve a:  
   **Git** → **Push...** (o presiona `Ctrl + Shift + K`).
2. Haz clic en el botón azul **Push**.

### Opción 2: Usar un Token de GitHub (PAT) por Terminal
Si prefieres hacerlo por terminal, puedes usar un token de acceso:

1. Ve a GitHub: [github.com/settings/tokens](https://github.com/settings/tokens) (o Settings → Developer settings → Personal access tokens → Tokens (classic)).
2. Haz clic en **Generate new token (classic)**, dale permisos marcando la casilla **`repo`**, y cópialo.
3. En la terminal ejecuta:
```powershell
git push https://TU_TOKEN_AQUI@github.com/Wallecabj12/seminario2026.git main
```
*(Reemplazando `TU_TOKEN_AQUI` por el token copiado).*

```kotlin
    private val soundMap = ConcurrentHashMap<Int, Int>() // padId -> soundId
    private val loadedSoundIds = ConcurrentHashMap.newKeySet<Int>()
    private val pendingPlayMap = ConcurrentHashMap<Int, Pad>() // soundId -> Pad
    private val activeStreams = ConcurrentHashMap<Int, Int>() // padId -> streamId

    init {
        soundPool.setOnLoadCompleteListener { _, sampleId, status ->
            if (status == 0) {
                loadedSoundIds.add(sampleId)
                pendingPlayMap.remove(sampleId)?.let { pad ->
                    play(pad)
                }
            }
        }
    }

    override fun isSoundLoaded(padId: Int): Boolean {
        val soundId = soundMap[padId] ?: return false
        return loadedSoundIds.contains(soundId)
    }

    override fun play(pad: Pad) {
        val soundId = soundMap[pad.id]
        if (soundId == null || !loadedSoundIds.contains(soundId)) {
            soundId?.let { pendingPlayMap[it] = pad }
            return
        }
        val streamId = soundPool.play(
            soundId,
            pad.volume,
            pad.volume,
            1,
            if (pad.isLoop) -1 else 0,
            pad.rate,
        )
        if (streamId != 0) {
            activeStreams[pad.id] = streamId
        }
    }
```

```kotlin
    fun createWavBytes(sampleRate: Int = SAMPLE_RATE, samples: ShortArray): ByteArray {
        val totalAudioLen = samples.size * 2
        val totalDataLen = totalAudioLen + 36
        val byteRate = sampleRate * 2

        val buffer = ByteBuffer.allocate(44 + totalAudioLen).order(ByteOrder.LITTLE_ENDIAN)
        buffer.put('R'.code.toByte())
        buffer.put('I'.code.toByte())
        buffer.put('F'.code.toByte())
        buffer.put('F'.code.toByte())
        buffer.putInt(totalDataLen)
        buffer.put('W'.code.toByte())
        buffer.put('A'.code.toByte())
        buffer.put('V'.code.toByte())
        buffer.put('E'.code.toByte())

        buffer.put('f'.code.toByte())
        buffer.put('m'.code.toByte())
        buffer.put('t'.code.toByte())
        buffer.put(' '.code.toByte())
        buffer.putInt(16) // Subchunk1Size
        buffer.putShort(1.toShort()) // PCM
        buffer.putShort(1.toShort()) // Mono
        buffer.putInt(sampleRate)
        buffer.putInt(byteRate)
        buffer.putShort(2.toShort()) // Block align
        buffer.putShort(16.toShort()) // Bits per sample

        buffer.put('d'.code.toByte())
        buffer.put('a'.code.toByte())
        buffer.put('t'.code.toByte())
        buffer.put('a'.code.toByte())
        buffer.putInt(totalAudioLen)

        for (sample in samples) {
            buffer.putShort(sample)
        }
        return buffer.array()
    }
```

```kotlin
package com.geovanivargas.planetamusica.domain.sequencer

data class StepPattern(
    val padId: Int,
    val steps: BooleanArray = BooleanArray(16),
    val isMuted: Boolean = false,
    val volume: Float = 1.0f,
) {
    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is StepPattern) return false
        return padId == other.padId &&
            steps.contentEquals(other.steps) &&
            isMuted == other.isMuted &&
            volume == other.volume
    }

    override fun hashCode(): Int {
        var result = padId
        result = 31 * result + steps.contentHashCode()
        result = 31 * result + isMuted.hashCode()
        result = 31 * result + volume.hashCode()
        return result
    }

    companion object {
        fun createDefaultPatterns(availablePadIds: List<Int>): List<StepPattern> {
            return availablePadIds.map { padId ->
                val steps = BooleanArray(16)
                when (padId) {
                    1 -> { // Kick: 1, 5, 9, 13
                        steps[0] = true
                        steps[4] = true
                        steps[8] = true
                        steps[12] = true
                    }
                    2 -> { // Snare: 5, 13
                        steps[4] = true
                        steps[12] = true
                    }
                    3 -> { // Hi-Hat cerrado: cada 2 pasos
                        for (i in 0 until 16 step 2) {
                            steps[i] = true
                        }
                    }
                }
                StepPattern(padId = padId, steps = steps)
            }
        }

        fun encodeToString(patterns: List<StepPattern>): String {
            return patterns.joinToString(";") { p ->
                val bits = p.steps.joinToString("") { if (it) "1" else "0" }
                "${p.padId}:$bits:${if (p.isMuted) 1 else 0}:${p.volume}"
            }
        }

        fun decodeFromString(encoded: String): List<StepPattern> {
            if (encoded.isBlank()) return emptyList()
            return encoded.split(";").mapNotNull { token ->
                val parts = token.split(":")
                if (parts.size >= 4) {
                    val padId = parts[0].toIntOrNull() ?: return@mapNotNull null
                    val bits = parts[1]
                    val steps = BooleanArray(16) { i -> bits.getOrNull(i) == '1' }
                    val isMuted = parts[2] == "1"
                    val volume = parts[3].toFloatOrNull() ?: 1.0f
                    StepPattern(padId = padId, steps = steps, isMuted = isMuted, volume = volume)
                } else null
            }
        }
    }
}
```

```kotlin
    private fun startLoop() {
        loopJob?.cancel()
        loopJob = scope.launch {
            var step = _currentStep.value
            var nextStepTime = timeProvider()

            while (isActive && _isPlaying.value) {
                val activePads = patterns.filter { !it.isMuted && it.steps.getOrElse(step) { false } }
                    .map { it.padId }

                _currentStep.value = step
                onStepListener?.invoke(step, activePads)

                val stepIntervalMs = (60_000.0 / (_bpm.value * 4.0)).toLong().coerceAtLeast(10L)
                nextStepTime += stepIntervalMs

                val now = timeProvider()
                val delayMs = (nextStepTime - now).coerceAtLeast(1L)

                delay(delayMs)

                if (timeProvider() - nextStepTime > stepIntervalMs * 2) {
                    nextStepTime = timeProvider()
                }

                step = (step + 1) % 16
            }
        }
```

```kotlin
@Composable
fun DawKnob(
    value: Float,
    onValueChange: (Float) -> Unit,
    modifier: Modifier = Modifier,
    label: String? = null,
    size: Dp = 36.dp,
    activeColor: Color = DawAccentOrange,
    trackColor: Color = DawDarkBackground,
    valueFormatter: (Float) -> String = { "${(it * 100).toInt()}%" },
) {
    var isDragging by remember { mutableStateOf(false) }
    var dragAccumulator by remember { mutableFloatStateOf(0f) }

    Box(
        modifier = modifier
            .sizeIn(minWidth = 48.dp, minHeight = 48.dp)
            .pointerInput(Unit) {
                detectVerticalDragGestures(
                    onDragStart = {
                        isDragging = true
                        dragAccumulator = value
                    },
                    onDragEnd = { isDragging = false },
                    onDragCancel = { isDragging = false },
                    onVerticalDrag = { change, dragAmount ->
                        change.consume()
                        val delta = -dragAmount / 180f
                        dragAccumulator = (dragAccumulator + delta).coerceIn(0f, 1f)
                        onValueChange(dragAccumulator)
                    },
                )
            },
        contentAlignment = Alignment.Center,
    ) {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            Box(contentAlignment = Alignment.Center) {
                Canvas(modifier = Modifier.size(size)) {
                    val strokeWidth = 3.dp.toPx()
                    val arcSize = Size(this.size.width - strokeWidth, this.size.height - strokeWidth)
                    val topLeft = Offset(strokeWidth / 2, strokeWidth / 2)

                    val startAngle = 135f
                    val sweepAngle = 270f

                    drawArc(
                        color = trackColor,
                        startAngle = startAngle,
                        sweepAngle = sweepAngle,
                        useCenter = false,
                        topLeft = topLeft,
                        size = arcSize,
                        style = Stroke(width = strokeWidth, cap = StrokeCap.Round),
                    )

                    drawArc(
                        color = activeColor,
                        startAngle = startAngle,
                        sweepAngle = sweepAngle * value.coerceIn(0f, 1f),
                        useCenter = false,
                        topLeft = topLeft,
                        size = arcSize,
                        style = Stroke(width = strokeWidth, cap = StrokeCap.Round),
                    )

                    val center = Offset(this.size.width / 2, this.size.height / 2)
                    val innerRadius = (this.size.width / 2) - strokeWidth * 1.5f

                    drawCircle(
                        brush = Brush.radialGradient(
                            colors = listOf(Color(0xFF3A424C), Color(0xFF1E2328)),
                            center = center,
                            radius = innerRadius,
                        ),
                        radius = innerRadius,
                        center = center,
                    )

                    val currentAngleRad = Math.toRadians((startAngle + sweepAngle * value.coerceIn(0f, 1f)).toDouble())
                    val indicatorInner = Offset(
                        x = (center.x + (innerRadius * 0.35f) * cos(currentAngleRad)).toFloat(),
                        y = (center.y + (innerRadius * 0.35f) * sin(currentAngleRad)).toFloat(),
                    )
                    val indicatorOuter = Offset(
                        x = (center.x + (innerRadius * 0.85f) * cos(currentAngleRad)).toFloat(),
                        y = (center.y + (innerRadius * 0.85f) * sin(currentAngleRad)).toFloat(),
                    )

                    drawLine(
                        color = if (value > 0f) activeColor else Color.Gray,
                        start = indicatorInner,
                        end = indicatorOuter,
                        strokeWidth = 2.dp.toPx(),
                        cap = StrokeCap.Round,
                    )
                }
            }

            if (label != null) {
                Spacer(Modifier.height(2.dp))
                Text(
                    text = label.uppercase(),
                    style = MaterialTheme.typography.labelSmall.copy(
                        fontSize = 9.sp,
                        fontWeight = FontWeight.Bold,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    ),
                )
            }
        }

        if (isDragging) {
            Box(
                modifier = Modifier
                    .align(Alignment.TopCenter)
                    .clip(RoundedCornerShape(3.dp))
                    .background(Color(0xFF121518))
                    .border(1.dp, DawAccentOrange, RoundedCornerShape(3.dp))
                    .padding(horizontal = 4.dp, vertical = 2.dp),
            ) {
                Text(
                    text = valueFormatter(value),
                    style = MaterialTheme.typography.labelSmall.copy(
                        color = DawAccentOrange,
                        fontFamily = FontFamily.Monospace,
                        fontSize = 9.sp,
                        fontWeight = FontWeight.Bold,
                    ),
                )
            }
        }
    }
}
```

```kotlin
import com.geovanivargas.planetamusica.ui.theme.DawLedGreen
import com.geovanivargas.planetamusica.ui.theme.DawLedRed
import com.geovanivargas.planetamusica.ui.theme.DawPanelBorder

@Composable
fun LevelMeter(
    level: Float,
    modifier: Modifier = Modifier,
    segments: Int = 12,
    width: Dp = 10.dp,
    height: Dp = 44.dp,
) {
    val shape = RoundedCornerShape(2.dp)

    Box(
        modifier = modifier
            .width(width)
            .height(height)
            .clip(shape)
            .background(DawLcdBackground)
            .border(1.dp, DawPanelBorder, shape),
    ) {
        Canvas(modifier = Modifier.size(width, height)) {
            val totalSegments = segments.coerceAtLeast(4)
            val segmentSpacing = 1.5.dp.toPx()
            val availableHeight = this.size.height - (totalSegments + 1) * segmentSpacing
            val segmentHeight = availableHeight / totalSegments
            val segmentWidth = this.size.width - 4.dp.toPx()
            val left = 2.dp.toPx()

            val activeCount = (level.coerceIn(0f, 1f) * totalSegments).toInt()

            for (i in 0 until totalSegments) {
                val indexFromBottom = i
                val isActive = indexFromBottom < activeCount

                val top = this.size.height - ((i + 1) * segmentHeight + (i + 1) * segmentSpacing)

                val ratio = i.toFloat() / totalSegments.toFloat()
                val segmentColor = when {
                    ratio >= 0.85f -> DawLedRed
                    ratio >= 0.65f -> DawAccentOrange
                    else -> DawLedGreen
                }

                drawRoundRect(
                    color = if (isActive) segmentColor else segmentColor.copy(alpha = 0.15f),
                    topLeft = Offset(left, top),
                    size = Size(segmentWidth, segmentHeight),
                    cornerRadius = CornerRadius(1.dp.toPx()),
                )
            }
        }
    }
}
```

### [SynthEngineTest.kt](file:///c:/Users/geova/AndroidStudioProjects/PlanetaMusica/app/src/test/java/com/geovanivargas/planetamusica/SynthEngineTest.kt)
```kotlin
package com.geovanivargas.planetamusica

import com.geovanivargas.planetamusica.audio.synth.*
import org.junit.Assert.assertTrue
import org.junit.Test
import kotlin.math.PI
import kotlin.math.sin
import kotlin.math.sqrt

class SynthEngineTest {

    @Test
    fun render_returnsCorrectDurationAndValidSampleRangeWithoutClipping() {
        val engine = SynthEngine()
        val durationMs = 500L
        val expectedMinSamples = ((durationMs / 1000.0) * SynthEngine.SAMPLE_RATE).toInt()

        val params = SynthParams()
        val samples = engine.render(params, midiNote = 60, durationMs = durationMs)

        assertTrue(samples.size >= expectedMinSamples)

        var hasNonZero = false
        var maxPeak = 0
        for (sample in samples) {
            val sInt = sample.toInt()
            assertTrue(sInt in -32768..32767)
            if (kotlin.math.abs(sInt) > maxPeak) maxPeak = kotlin.math.abs(sInt)
            if (sInt != 0) hasNonZero = true
        }

        assertTrue(hasNonZero)
        assertTrue(maxPeak <= 32767)
    }

    @Test
    fun filter_cutoffAttenuatesHighFrequencies() {
        val sampleRate = 44100
        val highFreq = 4000.0
        val numSamples = 1000

        val highFreqSignal = FloatArray(numSamples) { i ->
            sin(2.0 * PI * highFreq * i / sampleRate).toFloat()
        }

        val samplesLowCutoff = highFreqSignal.clone()
        val filterLow = BiquadFilter()
        filterLow.processBlock(samplesLowCutoff, FilterType.LOW_PASS, 500f, 0.1f, 0f, sampleRate)

        val samplesHighCutoff = highFreqSignal.clone()
        val filterHigh = BiquadFilter()
        filterHigh.processBlock(samplesHighCutoff, FilterType.LOW_PASS, 12000f, 0.1f, 0f, sampleRate)

        val rmsLow = sqrt(samplesLowCutoff.map { it * it }.average())
        val rmsHigh = sqrt(samplesHighCutoff.map { it * it }.average())

        assertTrue(rmsLow < rmsHigh * 0.35)
    }
}
```
